```cpp
#include <iostream>
#include <queue>
#include <thread>
#include <mutex>
#include <condition_variable>
#include <future>
#include <functional>
#include <vector>

// 任务队列：存储“带返回值的任务”（包装成 std::packaged_task）
using Task = std::function<void()>; // 包装 packaged_task 的无参函数

class ThreadPool {
private:
    std::queue<Task> task_queue_;   // 任务队列, 存储待执行任务，每个任务会设置对应的 future 结果，包装成无参函数，为了统一处理
    std::mutex mtx_;
    std::condition_variable cv_;    // 条件变量，通知工作线程有新任务，等待任务或停止信号
    std::vector<std::thread> workers_; // 工作线程池，消费者线程，负责取任务执行
    bool stop_ = false; // 停止标志，用于析构时停止线程池

public:
    ThreadPool(size_t n) { // n：工作线程数量
    for (size_t i = 0; i < n; ++i) { // 循环创建n个线程
        workers_.emplace_back([this]() { // 每个线程的执行逻辑（lambda表达式）
            while (true) { // 线程核心循环：持续处理任务
                Task task; // 存储取出的任务（Task = std::function<void()>）
                { // 代码块：限制互斥锁的作用域（自动释放锁）
                    // 1. 加锁 + 等待条件（任务队列非空 或 线程池停止）
                    std::unique_lock<std::mutex> lock(mtx_);
                    cv_.wait(lock, [this]() { return stop_ || !task_queue_.empty(); });

                    // 2. 退出条件：线程池停止 + 任务队列空（无任务可执行）
                    if (stop_ && task_queue_.empty()) return;

                    // 3. 从队列取出任务（移动语义，避免拷贝）
                    task = std::move(task_queue_.front());
                    task_queue_.pop();
                } // 锁自动释放（unique_lock析构）

                // 4. 执行任务（无锁，并发执行）
                task(); 
            }
        });
    }
}

    // 提交任务，返回 future（供主线程获取结果）
    template <typename F, typename... Args>
    auto submit(F&& f, Args&&... args) -> std::future<decltype(f(args...))> {
        // 包装任务为 packaged_task（关联 future）
        using ReturnType = decltype(f(args...));
        auto task = std::make_shared<std::packaged_task<ReturnType()>>(
            std::bind(std::forward<F>(f), std::forward<Args>(args)...)
        );
        std::future<ReturnType> fut = task->get_future();

        // 把 packaged_task 放入队列（包装成无参 Task）
        {
            std::lock_guard<std::mutex> lock(mtx_);
            task_queue_.emplace([task]() { (*task)(); });
        }
        cv_.notify_one(); // 唤醒工作线程

        return fut;
    }

    // 析构：停止线程池
    ~ThreadPool() {
        {
            std::lock_guard<std::mutex> lock(mtx_);
            stop_ = true; // 设置停止标志，通知所有线程退出，准备关闭线程池
        }
        cv_.notify_all();   // 唤醒所有线程，准备退出
        for (auto& worker : workers_) {
            worker.join();  // 等待所有线程结束
        }
    }
};

// 示例任务：计算平方
int square(int x) {
    std::this_thread::sleep_for(std::chrono::milliseconds(200));
    return x * x;
}

int main() {
    ThreadPool pool(2); // 2 个工作线程（消费者）

    // 提交 3 个任务（生产者），获取 future
    std::future<int> fut1 = pool.submit(square, 2);
    std::future<int> fut2 = pool.submit(square, 3);
    std::future<int> fut3 = pool.submit(square, 4);

    // 获取每个任务的结果（future 的核心作用）
    std::cout << "2² = " << fut1.get() << std::endl;
    std::cout << "3² = " << fut2.get() << std::endl;
    std::cout << "4² = " << fut3.get() << std::endl;

    return 0;
}
```
